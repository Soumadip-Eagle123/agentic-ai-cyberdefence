# Layer-to-layer contracts in src/schemas/contracts.py, and every layer that
# validates against them.

# FIPS 204 ML-DSA checkpoint signing in src/compliance/signing.py. Needs a
# build linked against OpenSSL 3.5+ for ML-DSA support; without it Layer 6
# falls back to HMAC checkpoints and four signing tests skip rather than fail.
cryptography>=46.0
